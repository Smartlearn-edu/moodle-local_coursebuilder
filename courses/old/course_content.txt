Google-style search query classifier:
# Search query safety classifier (simplified Google-style)
BANNED_WORDS = ["spam", "hack", "cheat"]

def classify_query(query):
    query_lower = query.lower()
    for word in BANNED_WORDS:
        if word in query_lower:
            return "BLOCKED"
    if len(query_lower) < 3:
        return "TOO_SHORT"
    elif len(query_lower) > 200:
        return "TOO_LONG"
    else:
        return "SAFE"

queries = ["Python tutorial", "hack wifi", "ai", "how to learn machine learning"]
for q in queries:
    print(f"  '{q[:30]}' -> {classify_query(q)}")

📘  CORE CONCEPTS     ⏱ 30 minutes
Netflix’s recommendation system — responsible for 80% of content watched — is built as hundreds of small, specialised functions. Each does one job: score a title, filter by language, apply a taste cluster. Functions are the building blocks of every major tech product.

Key Concepts:
def function_name(params): ... return value
Parameters vs arguments  |  default values  |  *args  |  **kwargs
return vs print — functions should return, not print
Scope: LEGB rule — Local, Enclosing, Global, Built-in
Docstrings: triple-quoted explanation at the top of every function
Lambda: score = lambda x: x * 1.5
Modules: import math, random, datetime, os

Netflix-style recommendation scorer:
import random

def calculate_match_score(user_rating, genre_weight=1.0, recency_bonus=0):
    """
    Calculate a Netflix-style match percentage for a title.
    Args:
        user_rating (float): Historical rating 0-5
        genre_weight (float): How much user likes this genre (0.5-2.0)
        recency_bonus (int): Extra points for recent titles
    Returns:
        int: Match percentage 0-100
    """
    base_score = (user_rating / 5.0) * 100
    weighted   = base_score * genre_weight + recency_bonus
    return min(int(weighted), 100)

titles = ["Stranger Things", "The Crown", "Squid Game"]
for title in titles:
    score = calculate_match_score(
        user_rating=random.uniform(3, 5),
        genre_weight=random.choice([0.8, 1.0, 1.5]),
        recency_bonus=random.randint(0, 10)
    )
    print(f"{title:20} -> {score}% Match")

🎯 Learning Outcomes

📘  CORE CONCEPTS     ⏱ 30 minutes
Spotify manages 100 million songs, 600 million users, and 5 billion playlists. Every playlist is a list. Every user profile is a dictionary. Artist genres are sets. Track metadata is a tuple. Picking the right structure is one of the most important engineering decisions at Spotify.

Key Concepts:
Lists []: ordered, mutable, duplicates allowed — .append() .remove() .sort() .pop()
Dicts {}: key–value, fast lookup — .get() .keys() .values() .items()
Tuples (): ordered, immutable — used for fixed data like (lat, lon)
Sets {}: unordered, unique only — .add() .union() .intersection() .difference()
Nested: list of dicts, dict of lists
Dict comprehensions: {k: v for k, v in items}

Spotify playlist — all 4 data structures:
# Spotify-style data structures
playlist = {
    "name":   "Morning Boost",
    "owner":  "alex_user",

        {"title": "Blinding Lights", "artist": "The Weeknd", "duration_s": 200},

print(f"Duration: {total_s // 60}m {total_s % 60}s")

Key Concepts:

Meta-style post logger with full error handling:
import csv, os
from datetime import datetime

def log_post(user_id, content, filename="posts.csv"):
    """Log a social post to a CSV file safely."""
    if not content.strip():
        raise ValueError("Post content cannot be empty")
    if len(content) > 280:
        raise ValueError(f"Post too long: {len(content)} chars (max 280)")
    try:
        file_exists = os.path.exists(filename)
        with open(filename, "a", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=["timestamp","user_id","content","chars"])
            if not file_exists:
                writer.writeheader()
            writer.writerow({
                "timestamp": datetime.now().isoformat(),
                "user_id":   user_id,
                "content":   content,
                "chars":     len(content)
            })
        print(f"Post logged for user {user_id}")
    except IOError as e:
        print(f"File error: {e}")

log_post("user_42", "Just learned Python file handling!")
log_post("user_99", "Building the next Facebook with Python!")

from datetime import datetime
import random

class Vehicle:
    def __init__(self, vehicle_id, make, model, capacity=4):
        self.vehicle_id = vehicle_id
        self.make       = make
        self.model      = model
        self.capacity   = capacity
    def __str__(self):
        return f"{self.make} {self.model} (ID: {self.vehicle_id})"

class Driver:
    def __init__(self, driver_id, name, rating=5.0):
        self.driver_id = driver_id
        self.name      = name
        self.rating    = rating
        self.trips     = []
    def accept_trip(self, trip):
        self.trips.append(trip)
        print(f"Driver {self.name} accepted trip #{trip.trip_id}")

class Trip:
    def __init__(self, rider_name, pickup, dropoff, distance_km):
        self.trip_id    = random.randint(1000, 9999)
        self.rider_name = rider_name
        self.pickup     = pickup
        self.dropoff    = dropoff
        self.distance   = distance_km
        self.fare       = round(2.50 + 1.20 * distance_km, 2)
        self.status     = "REQUESTED"
    def start(self):
        self.status = "IN_PROGRESS"
        print(f"Trip #{self.trip_id}: {self.pickup} to {self.dropoff}")
    def complete(self):
        self.status = "COMPLETED"
        print(f"Trip #{self.trip_id} done. Fare: ${self.fare}")

driver = Driver("D001", "Maria")
trip   = Trip("Alex", "Airport", "Downtown", 12.5)
driver.accept_trip(trip)
trip.start()
trip.complete()

import requests
import pandas as pd
import matplotlib.pyplot as plt

def fetch_top_python_repos():
    url    = "https://api.github.com/search/repositories"
    params = {"q": "language:python", "sort": "stars", "per_page": 10}
    response = requests.get(url, params=params)
    response.raise_for_status()
    return response.json()["items"]

repos = fetch_top_python_repos()
df = pd.DataFrame([{
    "name":  r["name"],
    "stars": r["stargazers_count"],
    "forks": r["forks_count"],
    "owner": r["owner"]["login"]
} for r in repos])

print(df[["name","stars","forks"]].to_string(index=False))

df.plot(x="name", y="stars", kind="barh", figsize=(10,6),
        title="Top 10 Python GitHub Repos by Stars", color="#1A73E8")
plt.tight_layout()
plt.savefig("top_repos.png")
print("Chart saved as top_repos.png")
